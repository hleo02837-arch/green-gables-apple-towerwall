    function buildSummary() {
      const client = (document.getElementById("client-name").value || "Client").trim();
      const macs = Math.max(1, parseInt(document.getElementById("mac-count").value, 10) || 1);
      const rate = Math.max(0, parseFloat(document.getElementById("rate-cad").value) || 0);
      const mode = document.getElementById("service-mode").value;
      const total = (macs * rate).toFixed(2);

      const text = [
        "GREEN GABLES — APPLE TOWERWALL | BOOKING & PAYMENT SUMMARY",
        `Client: ${client}`,
        `Service Mode: ${mode}`,
        `Macs Covered: ${macs}`,
        `Total Amount: $${total} CAD`,
        "Payment Method: Interac e-Transfer (Autodeposit)",
        "Send To: leorooflife8@gmail.com",
        `Memo Note: Towerwall Audit - ${client}`
      ].join("\n");

      document.getElementById("summary-out").textContent = text;
      return text;
    }

    document.getElementById("gen-btn").addEventListener("click", buildSummary);
    document.getElementById("copy-btn").addEventListener("click", async () => {

      }
    });
